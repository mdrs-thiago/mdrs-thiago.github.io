class AttnPool(nn.Module):
    """Atenção aditiva com uma query aprendida."""
    def __init__(self, d):
        super().__init__()
        self.proj = nn.Linear(d, d)
        self.q = nn.Parameter(torch.randn(d) / d**0.5)

    def forward(self, H, mask):      # H: (B, T, d)
        scores = torch.tanh(self.proj(H)) @ self.q  # (B, T)
        scores = scores.masked_fill(~mask, -1e9)    # ignora pads
        alpha = torch.softmax(scores, dim=1)        # (B, T)
        ctx = (alpha.unsqueeze(-1) * H).sum(dim=1)  # (B, d)
        return ctx, alpha
