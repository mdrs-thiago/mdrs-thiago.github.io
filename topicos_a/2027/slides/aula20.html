# latência de inferência por exemplo (CPU e GPU)
def latency(model, texts, device, n=50):
    model.to(device).eval()
    enc = tok(texts[:1], return_tensors="pt", truncation=True, max_length=256).to(device)
    with torch.no_grad():
        for _ in range(5): model(**enc)           # aquecimento
        t0 = time.time()
        for _ in range(n): model(**enc)
    return (time.time() - t0) / n * 1000          # ms

Acurácia não é tudo: em produção, latência e custo por requisição decidem. Um modelo 2 pontos melhor e 50× mais lento pode não valer a pena.
